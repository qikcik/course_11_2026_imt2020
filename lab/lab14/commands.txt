### 2. Ruch

# 20 UE ze stałą sesją PDU i tunelem (UE 1–20, każdy na własnym gNB)
sudo ~/5gs/PacketRusher/packetrusher --tunnel-backend userspace --config ~/5gs/repo/lab/lab14/config/packetrusher_ue.yml \
     multi-ue -n 20 -d --tunnel
ip -br link | grep -c vrf      # 20: tunele wszystkich UE są gotowe

# ruch z każdego z 20 UE: ping 10 razy na sekundę po 1200 bajtów, około 100 kbit/s w Uplinku i tyle samo w Downlinku
printf 'vrf%010d\n' $(seq 1 20) | sudo xargs -P 20 -I{} ip vrf exec {} ping -q -i 0.1 -s 1200 10.45.0.1
# większy ruch: -i 0.05 (około 200 kbit/s na UE)

# wariant zamiast dwóch poleceń powyżej: 20 UE z ruchem, które po awarii same wracają
# gdy ping z UE 1 przestaje działać, PacketRusher i pingi startują od nowa
while true; do
  sudo ~/5gs/PacketRusher/packetrusher --tunnel-backend userspace --config ~/5gs/repo/lab/lab14/config/packetrusher_ue.yml multi-ue -n 20 -d --tunnel &
  for t in $(seq 60); do [ "$(ip -br link | grep -c vrf)" = 20 ] && break; sleep 1; done
  printf 'vrf%010d
' $(seq 1 20) | sudo xargs -P 20 -I{} ip vrf exec {} ping -q -i 0.1 -s 1200 10.45.0.1 &
  sleep 5
  while sudo ip vrf exec vrf0000000001 ping -c 3 -W 1 -q 10.45.0.1 > /dev/null; do sleep 2; done
  sudo pkill -f "ping -q -i 0.1"; sudo pkill -INT -f packetrusher_ue.yml; sleep 5
done
# zatrzymanie: Ctrl+C, potem
sudo pkill -f "ping -q -i 0.1"; sudo pkill -INT -f packetrusher_ue.yml

# ping z UE 1: kontrola na bieżąco, czy User Plane działa
sudo ip vrf exec vrf0000000001 ping -i 0.2 10.45.0.1

# UE 21–80 w pętli: rejestracja, sesja PDU, po 4 s wyrejestrowanie
sudo ~/5gs/PacketRusher/packetrusher --tunnel-backend userspace --metrics-addr 127.0.0.101:9095 --config ~/5gs/repo/lab/lab14/config/packetrusher_ruch.yml \
     multi-ue -n 60 --loop --td 4000 --tbrr 2000 --tr 100

# UE 81–130 w pętli z handoverem Xn: rejestracja, sesja PDU, po 3 s handover na drugi gNB, po 6 s wyrejestrowanie
sudo ~/5gs/PacketRusher/packetrusher --tunnel-backend userspace --metrics-addr 127.0.0.110:9095 --config ~/5gs/repo/lab/lab14/config/packetrusher_ho_xn.yml \
     multi-ue -n 50 --number-of-gnbs 2 --loop --xnh 3000 --td 6000 --tbrr 1000 --tr 200

# UE 131–180 w pętli z handoverem N2: ten sam cykl, handover przez AMF
sudo ~/5gs/PacketRusher/packetrusher --tunnel-backend userspace --metrics-addr 127.0.0.112:9095 --config ~/5gs/repo/lab/lab14/config/packetrusher_ho_n2.yml \
     multi-ue -n 50 --number-of-gnbs 2 --loop --ngh 3000 --td 6000 --tbrr 1000 --tr 200


### 3. Awarie (każda z poleceniem, które ją cofa)

# zatrzymanie funkcji
sudo systemctl stop open5gs-udmd
# cofnięcie
sudo systemctl start open5gs-udmd

# nagły koniec procesu
sudo pkill -9 open5gs-upfd
# cofnięcie: systemd sam uruchamia proces; kontrola i ręczny start
systemctl is-active open5gs-upfd
sudo systemctl start open5gs-upfd

# to samo dla każdej funkcji: open5gs-nrfd, -amfd, -smfd, -upfd, -pcfd, -udmd, -udrd, -ausfd, -nssfd, -bsfd oraz mongod

# utrata 30% pakietów między gNB a UPF (N3, GTP-U)
sudo iptables -I INPUT -p udp --dport 2152 -m statistic --mode random --probability 0.3 -j DROP
# cofnięcie
sudo iptables -D INPUT -p udp --dport 2152 -m statistic --mode random --probability 0.3 -j DROP

# blokada między SMF a UPF (N4, PFCP)
sudo iptables -I INPUT -p udp --dport 8805 -j DROP
# cofnięcie
sudo iptables -D INPUT -p udp --dport 8805 -j DROP

# blokada między gNB a AMF (N2, SCTP)
sudo iptables -I INPUT -p sctp -j DROP
# cofnięcie
sudo iptables -D INPUT -p sctp -j DROP

# co trzeci pakiet do SBI ginie (TCP 7777, wszystkie NF)
sudo iptables -I INPUT -p tcp --dport 7777 -m statistic --mode nth --every 3 --packet 0 -j DROP
# cofnięcie
sudo iptables -D INPUT -p tcp --dport 7777 -m statistic --mode nth --every 3 --packet 0 -j DROP

# opóźnienie 200 ms wiadomości między SMF a UPF (N4, PFCP)
sudo tc qdisc add dev lo root handle 1: prio bands 4
sudo tc qdisc add dev lo parent 1:4 handle 40: netem delay 200ms
sudo tc filter add dev lo parent 1: protocol ip u32 match ip dport 8805 0xffff flowid 1:4
# cofnięcie (usuwa opóźnienie i wszystkie filtry)
sudo tc qdisc del dev lo root

# to samo dla innych połączeń: po dwóch pierwszych poleceniach inny filtr
#   gNB i AMF (N2, SCTP):        sudo tc filter add dev lo parent 1: protocol ip u32 match ip dport 38412 0xffff flowid 1:4
#   gNB i UPF (N3, GTP-U):       sudo tc filter add dev lo parent 1: protocol ip u32 match ip dport 2152 0xffff flowid 1:4
#   żądania SBI między NF:       sudo tc filter add dev lo parent 1: protocol ip u32 match ip dport 7777 0xffff flowid 1:4
# inne opóźnienie albo utrata: netem delay 1000ms, netem delay 200ms 50ms, netem loss 10%
# kontrola: liczba opóźnionych pakietów
tc -s qdisc show dev lo

# kontrola: lista reguł powinna być pusta
sudo iptables -L INPUT -n --line-numbers
# cofnięcie wszystkich reguł naraz (usuwa wszystkie reguły INPUT, także spoza labu)
sudo iptables -F INPUT

# większe obciążenie: zatrzymać rejestracje w pętli (Ctrl+C) i uruchomić z krótszym cyklem, na przykład --td 1000 --tbrr 500
# cofnięcie: Ctrl+C i polecenie z sekcji 2
