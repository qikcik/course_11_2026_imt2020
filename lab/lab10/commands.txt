# lab10: kto dopuszcza slice: abonent, gNB i AMF
# Pliki: current/pcap/ core_all.pcapng
# Terminale: T1 tcpdump, T2 5GC i PacketRusher


### 0. Przygotowanie (raz po starcie maszyny; "File exists" oznacza, że już jest)

sudo ip addr add 127.0.0.100/8 dev lo      # gNB PacketRusher (N2, N3)

# 5GC i abonent bez zmian (lab0): AMF obsługuje slice SST 1, abonent ma domyślny slice SST 1
# próby różnią się tylko slice'em gNB (slicesupportlist) i slice'em w prośbie UE o sesję (snssai):
#   A: gNB SST 1,            UE SST 1
#   B: gNB SST 1 SD 000001,  UE SST 1 SD 000001
#   C: gNB SST 1,            UE SST 2


### 1. Czysty start

cd ~/5gs/repo/lab/lab10
sudo rm -rf current/pcap current/log current/core_log current/core_config
mkdir -p current/{pcap,log,core_log,core_config}

sudo systemctl stop open5gs-amfd open5gs-smfd open5gs-upfd open5gs-ausfd open5gs-udmd open5gs-udrd open5gs-pcfd open5gs-nssfd open5gs-bsfd open5gs-nrfd
sudo rm -f /var/log/open5gs/*.log*


### 2. Przebieg

# T1: nagranie (przed startem 5GC)
sudo tcpdump -i lo -n -w current/pcap/core_all.pcapng "tcp port 7777 or udp port 8805 or udp port 2152 or sctp"

# T2: start 5GC
step "1 start 5GC"
sudo systemctl start open5gs-nrfd
sleep 1
sudo systemctl start open5gs-udrd open5gs-udmd open5gs-ausfd open5gs-pcfd open5gs-nssfd open5gs-bsfd open5gs-upfd open5gs-smfd open5gs-amfd

# T2: trzy uruchomienia PacketRushera; każde ok. 10 s, potem Ctrl+C
step "2 A: gNB i UE w slice SST 1"
sudo ~/5gs/PacketRusher/packetrusher --config ./config/packetrusher_A_sst1.yml multi-ue -n 1 -d 2>&1 | tee current/log/2_A_sst1.log

step "3 B: gNB obsługuje tylko SST 1 SD 000001"
sudo ~/5gs/PacketRusher/packetrusher --config ./config/packetrusher_B_gnb_sd000001.yml multi-ue -n 1 -d 2>&1 | tee current/log/3_B_gnb_sd000001.log

step "4 C: UE prosi o sesję w SST 2"
sudo ~/5gs/PacketRusher/packetrusher --config ./config/packetrusher_C_sesja_sst2.yml multi-ue -n 1 -d 2>&1 | tee current/log/4_C_sesja_sst2.log

# T1: Ctrl+C
step "5 koniec"


### 3. Zapis

sudo cp -R /etc/open5gs/. current/core_config/
sudo cp -R /var/log/open5gs/. current/core_log/
sudo chown -R $USER: current
